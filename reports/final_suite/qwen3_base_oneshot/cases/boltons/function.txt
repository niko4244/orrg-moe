def isoparse(iso_str):
    """Parses the limited subset of `ISO8601-formatted time`_ strings as
    returned by :meth:`datetime.datetime.isoformat`.

    >>> epoch_dt = datetime.fromtimestamp(0, timezone.utc).replace(tzinfo=None)
    >>> iso_str = epoch_dt.isoformat()
    >>> print(iso_str)
    1970-01-01T00:00:00
    >>> isoparse(iso_str)
    datetime.datetime(1970, 1, 1, 0, 0)

    >>> utcnow = datetime.now(timezone.utc).replace(tzinfo=None)
    >>> utcnow == isoparse(utcnow.isoformat())
    True

    For further datetime parsing, see the `iso8601`_ package for strict
    ISO parsing and `dateutil <https://pypi.python.org/pypi/python-dateutil>`__
    package for loose parsing and more.

    .. _ISO8601-formatted time: https://en.wikipedia.org/wiki/ISO_8601
    .. _iso8601: https://pypi.python.org/pypi/iso8601

    >>> isoparse('1970-01-01T00:00:00.851')
    datetime.datetime(1970, 1, 1, 0, 0, 0, 851000)

    Timezone suffixes (``Z``, ``+HH:MM``, ``-HHMM``, and so on) yield
    timezone-aware datetimes, so aware
    :meth:`~datetime.datetime.isoformat` output round-trips, too.

    >>> isoparse('1970-01-01T00:00:00+05:30')
    datetime.datetime(1970, 1, 1, 0, 0, tzinfo=datetime.timezone(datetime.timedelta(seconds=19800)))
    >>> aware_now = datetime.now(timezone.utc)
    >>> aware_now == isoparse(aware_now.isoformat())
    True

    """
    tzinfo = None
    tz_match = _ISO_TZ_RE.search(iso_str)
    if tz_match:
        iso_str = iso_str[:tz_match.start()]
        tz = tz_match.group(1)
        if tz == 'Z':
            tzinfo = timezone.utc
        else:
            digits = tz[1:].replace(':', '').split('.')[0]
            hh, mm = int(digits[0:2]), int(digits[2:4])
            ss = int(digits[4:6] or 0)
            sign = -1 if tz[0] == '-' else 1
            tzinfo = timezone(sign * timedelta(hours=hh, minutes=mm,
                                                seconds=ss))
    
    # Split the string by non-digits to extract date/time components
    parts = _NONDIGIT_RE.split(iso_str)
    dt_args = []
    for part in parts:
        if part == '':
            continue
        if part == '.':
            # Handle fractional seconds
            frac_part = part
            # Find the fractional part after the decimal point
            frac_match = _FRACTIONAL_RE.search(iso_str)
            if frac_match:
                frac_str = frac_match.group(0)
                frac_digits = frac_str.split('.')[1]
                # Convert fractional digits to microseconds
                frac_microseconds = int(frac_digits[:6]) if len(frac_digits) > 6 else int(frac_digits)
                frac_microseconds = frac_microseconds // 1000 if len(frac_digits) > 3 else frac_microseconds
                frac_microseconds = frac_microseconds if len(frac_digits) <= 6 else frac_microseconds % 1000000
                dt_args.append(0)  # Placeholder for microseconds
                # Append microseconds as a separate value
                if len(frac_digits) > 0:
                    frac_val = int(frac_digits[:6].ljust(6, '0')) // 1000
                    dt_args[-1] = frac_val
            else:
                dt_args.append(0)
        else:
            dt_args.append(int(part))
    
    # Reconstruct the datetime with proper microseconds
    # Extract the fractional part properly
    frac_part = None
    frac_digits = None
    for i, part in enumerate(parts):
        if part == '.':
            frac_part = i
            break
    
    if frac_part is not None:
        # Extract the fractional part after the decimal
        frac_str = iso_str.split('.')[1] if '.' in iso_str else ''
        frac_digits = frac_str[:6]  # Take up to 6 digits
        frac_microseconds = int(frac_digits.ljust(6, '0')) if frac_digits else 0
        frac_microseconds = frac_microseconds // 1000  # Convert to microseconds
        # Replace the last part with microseconds
        dt_args[-1] = frac_microseconds
    else:
        # No fractional part, just use the parts as is
        dt_args = [int(p) for p in parts]
    
    # Ensure we have the correct number of arguments
    if len(dt_args) < 7:
        # Pad with zeros if needed
        dt_args.extend([0] * (7 - len(dt_args)))
    
    return datetime(*dt_args, tzinfo=tzinfo)
